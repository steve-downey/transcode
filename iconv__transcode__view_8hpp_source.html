63// iconv_transcode_view<IconvFns, R>
64//
65// Adapts an input range R of legacy bytes into a byte-by-byte output range
66// in the target encoding, using an injected iconv-compatible callable set.
67// The iterator is intentionally move-only: iconv_t is an OS-managed opaque
68// handle that cannot be copied or shared.
69//
70// The caller provides an external output buffer (std::span<char>) to avoid
71// internal heap allocation.
97template <typename IconvFns, std::ranges::input_range R>
98 requires legacy_byte_range<R>
99class iconv_transcode_view : public std::ranges::view_interface<iconv_transcode_view<IconvFns, R>> {
101 R base_;
103 IconvFns fns_;
105 const char* from_;
107 const char* to_;
109 std::span<char> buffer_;
110
111 public:
114 class iterator {
115 using base_iter = std::ranges::iterator_t<R>;
116 using base_sent = std::ranges::sentinel_t<R>;
117
119 iconv_t handle_;
121 IconvFns fns_;
123 std::span<char> buffer_;
125 char* output_pos_;
127 char* output_end_;
128 // Accumulates unconsumed input bytes across load() calls so that
129 // multi-byte sequences can be assembled before passing to iconv.
131 char staging_[64];
133 size_t staging_len_{0};
135 base_iter current_;
137 base_sent end_;
139 bool done_;
141 detail::iconv_flush_state flush_state_{detail::iconv_flush_state::not_started};
143 bool stop_after_output_{false};
144
145 // Fills output_pos_/output_end_ with the next batch of converted bytes.
146 // Handles EINVAL (incomplete sequence) by accumulating more input, and
147 // E2BIG/EILSEQ by yielding partial output or skipping one staging byte.
148 // Sets done_ = true when all input and staging bytes are exhausted.
150 void load();
151
152 // Only iconv_transcode_view::begin() may construct an iterator.
153 friend class iconv_transcode_view;
155 iterator(iconv_t handle, IconvFns fns, std::span<char> buffer, base_iter current, base_sent end);
156
157 public:
158 using iterator_concept = std::input_iterator_tag;
159 using value_type = char;
160 using difference_type = std::ptrdiff_t;
161 using reference = char;
162
163 // \ref{transcode.iconv.iterator}, iterator operations
164 iterator(const iterator&) = delete;
165 iterator& operator=(const iterator&) = delete;
166
167 iterator(iterator&&) noexcept;
168 iterator& operator=(iterator&&) noexcept;
169 ~iterator();
170
173 const base_iter& base() const noexcept { return current_; }
174
175 char operator*() const;
176 iterator& operator++();
177 void operator++(int);
178
179 friend bool operator==(const iterator& it, std::default_sentinel_t) { return it.done_; }
180 };

249template <typename IconvFns, std::ranges::input_range R>
250 requires legacy_byte_range<R>
251void iconv_transcode_view<IconvFns, R>::iterator::load() {
252 if (stop_after_output_) {
253 done_ = true;
254 return;
255 }
256
257 char* out_ptr = buffer_.data();
258 size_t outleft = buffer_.size();
259
260 while (flush_state_ == detail::iconv_flush_state::not_started && outleft > 0) {
261 if constexpr (std::contiguous_iterator<base_iter> && std::sized_sentinel_for<base_sent, base_iter>) {
262 if (staging_len_ == 0 && current_ != end_) {
263 const auto* raw_ptr = reinterpret_cast<const char*>(std::to_address(current_));
264 auto remaining = static_cast<size_t>(end_ - current_);
265 char* in_ptr = const_cast<char*>(raw_ptr);
266 size_t inleft = remaining;
267 size_t rc = fns_.convert(handle_, &in_ptr, &inleft, &out_ptr, &outleft);
268 size_t consumed = remaining - inleft;
269 current_ += static_cast<std::ptrdiff_t>(consumed);
270 if (rc != iconv_error_rc) {
271 if (current_ == end_)
272 break;
273 continue;
274 }
275 if (errno == E2BIG) {
276 if (out_ptr == buffer_.data()) {
277 done_ = true;
278 return;
279 }
280 break;
281 }
282 if (errno == EINVAL) {
283 if (inleft > 0 && inleft <= sizeof(staging_)) {
284 std::copy_n(in_ptr, inleft, staging_);
285 staging_len_ = inleft;
286 current_ += static_cast<std::ptrdiff_t>(inleft);
287 }
288 break;
289 }
290 if (errno == EILSEQ) {
291 if (current_ != end_)
292 ++current_;
293 if (current_ == end_ && staging_len_ == 0)
294 break;
295 continue;
296 }
297 if (out_ptr != buffer_.data())
298 stop_after_output_ = true;
299 else
300 done_ = true;
301 break;
302 }
303 }
304
305 if (done_)
306 break;
307
308 // Slow path: fill staging from non-contiguous input or drain residue.
309 while (staging_len_ < sizeof(staging_) && current_ != end_) {
310 staging_[staging_len_++] = static_cast<char>(*current_);
311 ++current_;
312 }
313 if (staging_len_ == 0)
314 break;
315
316 char* in_ptr = staging_;
317 size_t inleft = staging_len_;
318 size_t rc = fns_.convert(handle_, &in_ptr, &inleft, &out_ptr, &outleft);
319 size_t consumed = staging_len_ - inleft;
320 if (consumed > 0 && inleft > 0)
321 std::copy_n(in_ptr, inleft, staging_);
322 staging_len_ = inleft;
323
324 if (rc != iconv_error_rc) {
325 if (staging_len_ == 0 && current_ == end_)
326 break;
327 continue;
328 }
329 if (errno == E2BIG) {
330 if (out_ptr == buffer_.data()) {
331 done_ = true;
332 return;
333 }
334 break;
335 }
336 if (errno == EINVAL) {
337 if (current_ == end_) {
338 staging_len_ = 0;
339 break;
340 }
341 continue;
342 }
343 if (errno == EILSEQ) {
344 if (staging_len_ > 0) {
345 std::copy_n(staging_ + 1, staging_len_ - 1, staging_);
346 --staging_len_;
347 }
348 if (staging_len_ == 0 && current_ == end_)
349 break;
350 continue;
351 }
352 if (out_ptr != buffer_.data())
353 stop_after_output_ = true;
354 else
355 done_ = true;
356 break;
357 }
358
359 output_pos_ = buffer_.data();
360 output_end_ = out_ptr;
361
362 if (output_pos_ != output_end_ || done_)
363 return;
364
365 if (flush_state_ == detail::iconv_flush_state::done) {
366 done_ = true;
367 return;
368 }
369
370 flush_state_ = detail::iconv_flush_state::in_progress;
371 char* flush_out = buffer_.data();
372 size_t flush_left = buffer_.size();
373 size_t rc = fns_.convert(handle_, nullptr, nullptr, &flush_out, &flush_left);
374 output_pos_ = buffer_.data();
375 output_end_ = flush_out;
376
377 if (rc != iconv_error_rc) {
378 flush_state_ = detail::iconv_flush_state::done;
379 if (output_pos_ == output_end_)
380 done_ = true;
381 return;
382 }
383
384 if (errno == E2BIG) {
385 if (output_pos_ == output_end_) {
386 flush_state_ = detail::iconv_flush_state::done;
387 done_ = true;
388 }
389 return;
390 }
391
392 flush_state_ = detail::iconv_flush_state::done;
393 if (output_pos_ != output_end_)
394 stop_after_output_ = true;
395 else
396 done_ = true;
397}

406iconv_transcode_view<IconvFns, R>::iterator::iterator(iterator&& other) noexcept

431 if (this != &other) {
432 if (handle_ != (iconv_t)-1)
433 fns_.close(handle_);
434 handle_ = other.handle_;
435 fns_ = std::move(other.fns_);
436 buffer_ = other.buffer_;
437 output_pos_ = other.output_pos_;
438 output_end_ = other.output_end_;
439 staging_len_ = other.staging_len_;
440 for (size_t i = 0; i < other.staging_len_; ++i)
441 staging_[i] = other.staging_[i];
442 current_ = std::move(other.current_);
443 end_ = std::move(other.end_);
444 done_ = other.done_;
445 flush_state_ = other.flush_state_;
446 stop_after_output_ = other.stop_after_output_;
447 other.handle_ = (iconv_t)-1;
448 other.done_ = true;
449 other.staging_len_ = 0;
450 }
451 return *this;
452}

494// iconv_transcode_closure<IconvFns> — pipe adapter for iconv_transcode_view.
495//
496// Stores the callable set, encoding pair, and output buffer so that
497// operator| can construct the view lazily:
498// auto v = input | iconv_transcode_closure<iconv_functions>{fns, "UTF-8", "UTF-32LE", buf};
500template <typename IconvFns>
501struct iconv_transcode_closure {
502 IconvFns fns_;
503 const char* from_;
504 const char* to_;
505 std::span<char> buffer_;
506
507 template <legacy_byte_range R>
508 auto operator()(R&& r) const {
509 return iconv_transcode_view<IconvFns, std::views::all_t<R>>(
510 std::views::all(std::forward<R>(r)), fns_, from_, to_, buffer_);
511 }
512
513 template <legacy_byte_range R>
514 friend auto operator|(R&& r, const iconv_transcode_closure& self) {
515 return self(std::forward<R>(r));
516 }
517};
