17// A WHATWG index is an array from pointer to code point. Decoding reads it
18// directly; encoding needs the other direction. WHATWG defines "the index
19// pointer for code point" as the first pointer whose entry is that code
20// point, and finding it by scanning the array costs O(N) per code point.
21// A reverse_index answers the same question in two loads.
22//
23// The reverse index is paged on the high bits of the code point: page_of
24// maps cp >> 8 to a page number, and each page maps the low 8 bits to
25// pointer + 1, with 0 meaning the code point is not in the index. Pages
26// that would be entirely empty are not stored, so the size is proportional
27// to the number of 256-code-point blocks the index touches, not to the code
28// space. Index jis0208 touches 98 of them: about 50 KiB.
29//
30// Some encoders exclude a range of pointers from the search. Shift_JIS
31// excludes 8272-8835, the NEC-selected IBM extensions, so that their code
32// points encode as the IBM extensions proper at 8836 and up. The excluded
33// range is applied when the index is built, so find() needs no special case.
34//
35// Build one with reverse_index_for<table, excluded>. The table must be a
36// constexpr array with static storage duration, which every generated index
37// in detail/tables/ is.
38
39// An index by address and length, so that it can be a template argument.
40// Converts implicitly from the array, so callers name the table itself. The
41// generated indexes are C arrays; the pluggable table codecs take std::array.
42struct table_ref {
43 const char32_t* data{nullptr};
44 std::size_t size{0};
45
46 template <std::size_t N>
47 constexpr table_ref(const char32_t (&table)[N]);
48
49 template <std::size_t N>
50 constexpr table_ref(const std::array<char32_t, N>& table);
51
52 constexpr table_ref(const char32_t* table, std::size_t count);
53};

55// An inclusive range of pointers; empty when last < first.
56struct pointer_range {
57 int first{0};
58 int last{-1};
59
60 constexpr bool contains(int pointer) const;
61};

63// The two sizes a reverse_index needs before it can be a type: how many page
64// slots the highest mapped code point reaches, and how many of those pages
65// hold at least one code point.
66struct reverse_index_shape {
67 std::size_t page_slots{0};
68 std::size_t page_count{0};
69};
70
71template <reverse_index_shape Shape>
72struct reverse_index {
73 static constexpr std::uint16_t no_page = 0xFFFF;
74
75 // Plain arrays, not std::array: the constant-evaluated build below touches
76 // these tens of thousands of times, and std::array's accessor is a function
77 // call at every touch, with an assertion in hardened library builds. The
78 // extra statements would put a large index over clang's default step
79 // limit. An empty table still needs one element to be a valid array.
80 std::uint16_t page_of[Shape.page_slots > 0 ? Shape.page_slots : 1]{};
81 std::uint16_t pages[Shape.page_count > 0 ? Shape.page_count : 1][256]{};
82
83 // The lowest pointer whose entry is cp, or -1 when cp is not in the index.
84 constexpr int find(char32_t cp) const;
85};

95// Holds the reverse index of Table as a static member, so that each
96// translation unit builds it once, at compile time.
97template <table_ref Table, pointer_range Excluded = {}>
98struct reverse_index_builder {
99 static_assert(Table.size < 0xFFFF, "transcode: reverse_index stores pointer + 1 in 16 bits");
100
101 static constexpr reverse_index_shape shape = reverse_index_shape_of(Table, Excluded);
102 static constexpr reverse_index<shape> value = make_reverse_index<shape>(Table, Excluded);
103};

105// The reverse index of Table: reverse_index_for<tables::shift_jis, pointer_range{8272, 8835}>.
106template <table_ref Table, pointer_range Excluded = {}>
107inline constexpr const auto& reverse_index_for = reverse_index_builder<Table, Excluded>::value;

123template <reverse_index_shape Shape>
124constexpr int reverse_index<Shape>::find(char32_t cp) const {
125 std::size_t hi = cp >> 8;
126 if (hi >= Shape.page_slots)
127 return -1;
128 std::uint16_t page = page_of[hi];
129 if (page == no_page)
130 return -1;
131 return static_cast<int>(pages[page][cp & 0xFF]) - 1;
132}

134constexpr reverse_index_shape reverse_index_shape_of(table_ref table, pointer_range excluded) {
135 // One pass: the highest code point and the set of pages it touches.
136 // Unicode has 0x1100 blocks of 256 code points.
137 bool seen[0x1100] = {};
138 reverse_index_shape shape{};
139 char32_t highest = 0;
140 const char32_t* const data = table.data;
141 for (std::size_t p = 0; p < table.size; ++p) {
142 const char32_t cp = data[p];
143 if (cp == 0)
144 continue;
145 const int pointer = static_cast<int>(p);
146 if (pointer >= excluded.first && pointer <= excluded.last)
147 continue;
148 highest = std::max(highest, cp);
149 if (!seen[cp >> 8]) {
150 seen[cp >> 8] = true;
151 ++shape.page_count;
152 }
153 }
154 if (shape.page_count > 0)
155 shape.page_slots = static_cast<std::size_t>(highest >> 8) + 1;
156 return shape;
157}

159template <reverse_index_shape Shape>
160constexpr reverse_index<Shape> make_reverse_index(table_ref table, pointer_range excluded) {
161 reverse_index<Shape> index{};
162 for (std::size_t hi = 0; hi < Shape.page_slots; ++hi)
163 index.page_of[hi] = reverse_index<Shape>::no_page;
164
165 // Touch every page entry once, in order, before the scattered writes below.
166 // GCC keeps a constant-evaluated array sparse until an element is written,
167 // and inserting elements out of order costs a shift of everything after
168 // them; filling densely first makes each later write an in-place update
169 // and cuts the build of a large index from about a second to a fifth of one.
170 for (std::size_t page = 0; page < Shape.page_count; ++page)
171 for (std::size_t lo = 0; lo < 256; ++lo)
172 index.pages[page][lo] = 0;
173
174 std::uint16_t next_page = 0;
175 const char32_t* const data = table.data;
176 for (std::size_t p = 0; p < table.size; ++p) {
177 const char32_t cp = data[p];
178 if (cp == 0)
179 continue;
180 const int pointer = static_cast<int>(p);
181 if (pointer >= excluded.first && pointer <= excluded.last)
182 continue;
183 const std::size_t hi = cp >> 8;
184 if (index.page_of[hi] == reverse_index<Shape>::no_page)
185 index.page_of[hi] = next_page++;
186 std::uint16_t& entry = index.pages[index.page_of[hi]][cp & 0xFF];
187 if (entry == 0)
188 entry = static_cast<std::uint16_t>(pointer + 1);
189 }
190 return index;
191}
