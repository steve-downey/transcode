63template <codec C, std::ranges::random_access_range R, transcode_error_kind E = transcode_error_kind::replacement>
64 requires unicode_scalar_range<R> && detail::random_access_encode_codec<C>
65class random_access_whatwg_encode_view
66 : public std::ranges::view_interface<random_access_whatwg_encode_view<C, R, E>> {
67 R base_;
68
69 class iterator {
70 using base_iter = detail::compatible_iterator_t<R>;
71
72 base_iter current_{};
73
74 public:
75 using iterator_concept = std::random_access_iterator_tag;
76 using iterator_category = std::random_access_iterator_tag;
77 using value_type =
78 std::conditional_t<E == transcode_error_kind::expected, std::expected<char, whatwg_error>, char>;
79
80 // A code point this codec cannot encode: the error itself under
81 // `expected`, and the '?' the lossy encoder substitutes otherwise.
82 static constexpr value_type error_result(whatwg_error e) {
83 if constexpr (E == transcode_error_kind::expected)
84 return value_type(std::unexpect, e);
85 else
86 return '?';
87 }
88 using difference_type = std::ptrdiff_t;
89 using reference = value_type;
90
91 constexpr iterator() = default;
92 constexpr explicit iterator(base_iter current) : current_(current) {}
93
94 constexpr const base_iter& base() const noexcept { return current_; }
95
96 constexpr value_type operator*() const {
97 const auto cp = static_cast<char32_t>(*current_);
98 if (cp < 0x80)
99 return static_cast<char>(cp);
100 const auto& table = detail::random_access_encode_table<C>();
101 for (int index = 0; index < 128; ++index) {
102 if (table[index] == cp)
103 return static_cast<char>(0x80 + index);
104 }
105 return error_result(whatwg_error::unmapped_codepoint);
106 }
107
108 constexpr value_type operator[](difference_type n) const { return *(*this + n); }
109
110 constexpr iterator& operator++() {
111 ++current_;
112 return *this;
113 }
114
115 constexpr iterator operator++(int) {
116 auto tmp = *this;
117 ++*this;
118 return tmp;
119 }
120
121 constexpr iterator& operator--() {
122 --current_;
123 return *this;
124 }
125
126 constexpr iterator operator--(int) {
127 auto tmp = *this;
128 --*this;
129 return tmp;
130 }
131
132 constexpr iterator& operator+=(difference_type n) {
133 current_ += n;
134 return *this;
135 }
136
137 constexpr iterator& operator-=(difference_type n) {
138 current_ -= n;
139 return *this;
140 }
141
142 constexpr iterator operator+(difference_type n) const {
143 auto tmp = *this;
144 return tmp += n;
145 }
146
147 constexpr iterator operator-(difference_type n) const {
148 auto tmp = *this;
149 return tmp -= n;
150 }
151
152 constexpr difference_type operator-(const iterator& other) const { return current_ - other.current_; }
153
154 constexpr friend iterator operator+(difference_type n, iterator it) { return it += n; }
155
156 constexpr auto operator<=>(const iterator&) const = default;
157 };
158
159 public:
160 constexpr explicit random_access_whatwg_encode_view(R base) : base_(std::move(base)) {}
161
162 constexpr const R& base() const& noexcept { return base_; }
163 constexpr R base() && { return std::move(base_); }
164
165 constexpr iterator begin() { return iterator(std::ranges::begin(base_)); }
166
167 constexpr iterator begin() const
168 requires detail::const_iterator_compatible_range<R>
169 {
170 return iterator(std::ranges::begin(base_));
171 }
172
173 constexpr iterator end() { return iterator(std::ranges::end(base_)); }
174
175 constexpr iterator end() const
176 requires detail::const_iterator_compatible_range<R> && std::ranges::range<const R>
177 {
178 return iterator(std::ranges::end(base_));
179 }
180
181 constexpr auto size() const
182 requires std::ranges::sized_range<R>
183 {
184 return std::ranges::size(base_);
185 }
186};

189// whatwg_encode_view — encodes char32_t codepoints to
190// expected<char, whatwg_error>, yielding unmapped_codepoint on failure.
191// ---------------------------------------------------------------------------
192
193template <codec C, std::ranges::input_range R, transcode_error_kind E = transcode_error_kind::replacement>
194 requires detail::whatwg_encode_input<C, R>
195class whatwg_encode_view : public std::ranges::view_interface<whatwg_encode_view<C, R, E>> {
196 R base_;
197
198 class iterator {
199 using base_iter = detail::compatible_iterator_t<R>;
200 using base_sent = detail::compatible_sentinel_t<R>;
201 using result_t =
202 std::conditional_t<E == transcode_error_kind::expected, std::expected<char, whatwg_error>, char>;
203
204 base_iter current_{};
205 base_sent end_{};
206 result_t buf_[8]{};
207 int len_{0};
208 int pos_{0};
209 bool done_{false};
210 int iso2022jp_state_{0};
211
212 // Report a code point this codec cannot encode: one error element under
213 // `expected`, and the codec's own replacement bytes otherwise -- which
214 // differ in length, so this cannot be a value mapping.
215 constexpr void emit_error(whatwg_error e, std::initializer_list<char> replacement) {
216 if constexpr (E == transcode_error_kind::expected) {
217 buf_[0] = result_t(std::unexpect, e);
218 len_ = 1;
219 } else {
220 int i = 0;
221 for (char c : replacement)
222 buf_[i++] = result_t(c);
223 len_ = i;
224 }
225 pos_ = 0;
226 }
227
228 constexpr void load();
229
230 public:
231 constexpr iterator() = default;
232
233 static constexpr iterator terminal()
234 requires std::ranges::forward_range<R>
235 {
236 iterator it;
237 it.done_ = true;
238 return it;
239 }
240
241 using iterator_concept =
242 std::conditional_t<std::ranges::forward_range<R>, std::forward_iterator_tag, std::input_iterator_tag>;
243 using iterator_category = iterator_concept;
244 using value_type = result_t;
245 using difference_type = std::ptrdiff_t;
246 using reference = result_t;
247
248 constexpr iterator(base_iter current, base_sent end);
249
250 constexpr const base_iter& base() const noexcept { return current_; }
251
252 constexpr auto operator*() const;
253 constexpr iterator& operator++();
254 constexpr iterator operator++(int)
255 requires std::ranges::forward_range<R>;
256 constexpr void operator++(int)
257 requires(!std::ranges::forward_range<R>);
258
259 constexpr friend bool operator==(const iterator& lhs, const iterator& rhs)
260 requires std::ranges::forward_range<R>
261 {
262 if (lhs.done_ || rhs.done_)
263 return lhs.done_ == rhs.done_;
264 return lhs.current_ == rhs.current_ && lhs.len_ == rhs.len_ && lhs.pos_ == rhs.pos_ &&
265 lhs.done_ == rhs.done_ && lhs.iso2022jp_state_ == rhs.iso2022jp_state_ &&
266 std::equal(std::begin(lhs.buf_), std::end(lhs.buf_), std::begin(rhs.buf_));
267 }
268
269 constexpr friend bool operator==(const iterator& it, std::default_sentinel_t) { return it.done_; }
270 };
271
272 public:
273 constexpr explicit whatwg_encode_view(R base);
274
275 constexpr const R& base() const& noexcept { return base_; }
276 constexpr R base() && { return std::move(base_); }
277
278 constexpr iterator begin();
279 constexpr iterator begin() const
280 requires detail::const_iterator_compatible_range<R> && detail::const_sentinel_compatible_range<R>;
281 constexpr iterator end()
282 requires std::ranges::forward_range<R> && std::ranges::common_range<R>;
283 constexpr iterator end() const
284 requires std::ranges::forward_range<const R> && std::ranges::common_range<const R> &&
285 detail::const_iterator_compatible_range<R> && detail::const_sentinel_compatible_range<R>;
286 constexpr std::default_sentinel_t end() const;
287};

289template <codec C, transcode_error_kind E = transcode_error_kind::replacement>
290struct whatwg_encode_closure {
291 template <typename R>
292 requires detail::whatwg_encode_input<C, R>
293 constexpr auto operator()(R&& r) const;
294
295 template <unicode_scalar_range R>
296 requires(!detail::whatwg_encode_codec<C>)
297 constexpr auto operator()(R&&) const {
298 static_assert(detail::whatwg_encode_codec<C>, "transcode: WHATWG does not define an encoder for this codec");
299 }
300
301 template <typename R>
302 requires detail::whatwg_encode_input<C, R>
303 constexpr friend auto operator|(R&& r, const whatwg_encode_closure& self) {
304 return self(std::forward<R>(r));
305 }
306
307 template <unicode_scalar_range R>
308 requires(!detail::whatwg_encode_codec<C>)
309 constexpr friend auto operator|(R&& r, const whatwg_encode_closure& self) {
310 return self(std::forward<R>(r));
311 }
312
313 template <typename R>
314 requires std::is_array_v<std::remove_cvref_t<R>>
315 friend auto operator|(R&&, const whatwg_encode_closure&) {
316 static_assert(!std::is_array_v<std::remove_cvref_t<R>>,
317 "transcode: raw arrays are not valid input to whatwg_encode; "
318 "wrap with std::span or std::vector");
319 }
320};

405template <codec C, std::ranges::input_range R, transcode_error_kind E>
406 requires detail::whatwg_encode_input<C, R>
407constexpr void whatwg_encode_view<C, R, E>::iterator::load() {
408 using result_value = std::remove_cvref_t<decltype(this->buf_[0])>;
409
410 if (current_ == end_) {
411 if constexpr (C == codec::iso_2022_jp) {
412 if (iso2022jp_state_ != 0) {
413 this->buf_[0] = result_value{'\x1B'};
414 this->buf_[1] = result_value{'\x28'};
415 this->buf_[2] = result_value{'\x42'};
416 len_ = 3;
417 pos_ = 0;
418 iso2022jp_state_ = 0;
419 return;
420 }
421 }
422 done_ = true;
423 return;
424 }
425 auto encode_single = [&](const char32_t (&table)[128]) {
426 auto r = detail::single_byte_encode_one(current_, end_, table);
427 if (r.is_error) {
428 this->emit_error(whatwg_error::unmapped_codepoint, {'?'});
429 } else {
430 this->buf_[0] = result_value{static_cast<char>(r.byte)};
431 len_ = 1;
432 pos_ = 0;
433 }
434 };
435 if constexpr (C == codec::utf_8) {
436 auto r = detail::utf8_encode_one(static_cast<char32_t>(*current_));
437 ++current_;
438 if (r.is_error) {
439 // U+FFFD in UTF-8
440 this->emit_error(r.error, {'\xEF', '\xBF', '\xBD'});
441 } else {
442 for (int i = 0; i < r.count; ++i)
443 this->buf_[i] = result_value{r.bytes[i]};
444 len_ = r.count;
445 }
446 pos_ = 0;
447 } else if constexpr (detail::random_access_encode_codec<C>) {
448 encode_single(detail::random_access_encode_table<C>());
449 } else if constexpr (C == codec::gbk) {
450 auto r = detail::gbk_encode_one(static_cast<char32_t>(*current_));
451 ++current_;
452 if (r.is_error) {
453 this->emit_error(whatwg_error::unmapped_codepoint, {'?'});
454 } else {
455 for (int i = 0; i < r.count; ++i)
456 this->buf_[i] = result_value{static_cast<char>(r.bytes[i])};
457 len_ = r.count;
458 }
459 pos_ = 0;
460 } else if constexpr (C == codec::gb18030) {
461 auto r = detail::gb18030_encode_one(static_cast<char32_t>(*current_));
462 ++current_;
463 if (r.is_error) {
464 this->emit_error(whatwg_error::unmapped_codepoint, {'?'});
465 } else {
466 for (int i = 0; i < r.count; ++i)
467 this->buf_[i] = result_value{static_cast<char>(r.bytes[i])};
468 len_ = r.count;
469 pos_ = 0;
470 }
471 } else if constexpr (C == codec::big5) {
472 auto r = detail::big5_encode_one(static_cast<char32_t>(*current_));
473 ++current_;
474 if (r.is_error) {
475 this->emit_error(whatwg_error::unmapped_codepoint, {'?'});
476 } else {
477 for (int i = 0; i < r.count; ++i)
478 this->buf_[i] = result_value{static_cast<char>(r.bytes[i])};
479 len_ = r.count;
480 }
481 pos_ = 0;
482 } else if constexpr (C == codec::shift_jis) {
483 auto r = detail::shift_jis_encode_one(static_cast<char32_t>(*current_));
484 ++current_;
485 if (r.is_error) {
486 this->emit_error(whatwg_error::unmapped_codepoint, {'?'});
487 } else {
488 for (int i = 0; i < r.count; ++i)
489 this->buf_[i] = result_value{static_cast<char>(r.bytes[i])};
490 len_ = r.count;
491 }
492 pos_ = 0;
493 } else if constexpr (C == codec::euc_jp) {
494 auto r = detail::euc_jp_encode_one(static_cast<char32_t>(*current_));
495 ++current_;
496 if (r.is_error) {
497 this->emit_error(whatwg_error::unmapped_codepoint, {'?'});
498 } else {
499 for (int i = 0; i < r.count; ++i)
500 this->buf_[i] = result_value{static_cast<char>(r.bytes[i])};
501 len_ = r.count;
502 }
503 pos_ = 0;
504 } else if constexpr (C == codec::iso_2022_jp) {
505 auto cp = static_cast<char32_t>(*current_);
506 ++current_;
507 // Roman state: U+00A5 (YEN SIGN) and U+203E (OVERLINE)
508 if (cp == 0x00A5 || cp == 0x203E) {
509 char ascii_byte = (cp == 0x00A5) ? '\x5C' : '\x7E';
510 if (iso2022jp_state_ != 1) {
511 this->buf_[0] = result_value{'\x1B'};
512 this->buf_[1] = result_value{'\x28'};
513 this->buf_[2] = result_value{'\x4A'};
514 this->buf_[3] = result_value{ascii_byte};
515 len_ = 4;
516 iso2022jp_state_ = 1;
517 } else {
518 this->buf_[0] = result_value{ascii_byte};
519 len_ = 1;
520 }
521 pos_ = 0;
522 return;
523 }
524 if (cp < 0x80) {
525 char ascii_byte = static_cast<char>(cp);
526 if (iso2022jp_state_ != 0) {
527 this->buf_[0] = result_value{'\x1B'};
528 this->buf_[1] = result_value{'\x28'};
529 this->buf_[2] = result_value{'\x42'};
530 this->buf_[3] = result_value{ascii_byte};
531 len_ = 4;
532 iso2022jp_state_ = 0;
533 } else {
534 this->buf_[0] = result_value{ascii_byte};
535 len_ = 1;
536 }
537 pos_ = 0;
538 return;
539 }
540 for (int i = 0; i < 8836; ++i) {
541 if (detail::tables::shift_jis[i] == cp) {
542 int lead = (i / 94) + 0x21;
543 int trail = (i % 94) + 0x21;
544 if (iso2022jp_state_ != 2) {
545 this->buf_[0] = result_value{'\x1B'};
546 this->buf_[1] = result_value{'\x24'};
547 this->buf_[2] = result_value{'\x42'};
548 this->buf_[3] = result_value{static_cast<char>(lead)};
549 this->buf_[4] = result_value{static_cast<char>(trail)};
550 len_ = 5;
551 iso2022jp_state_ = 2;
552 } else {
553 this->buf_[0] = result_value{static_cast<char>(lead)};
554 this->buf_[1] = result_value{static_cast<char>(trail)};
555 len_ = 2;
556 }
557 pos_ = 0;
558 return;
559 }
560 }
561 // Unmapped. The lossy encoder returns to ASCII before substituting, so
562 // the replacement carries the escape sequence with it.
563 if (iso2022jp_state_ != 0) {
564 iso2022jp_state_ = 0;
565 this->emit_error(whatwg_error::unmapped_codepoint, {'\x1B', '\x28', '\x42', '?'});
566 } else {
567 this->emit_error(whatwg_error::unmapped_codepoint, {'?'});
568 }
569 } else if constexpr (C == codec::euc_kr) {
570 auto r = detail::euc_kr_encode_one(static_cast<char32_t>(*current_));
571 ++current_;
572 if (r.is_error) {
573 this->emit_error(whatwg_error::unmapped_codepoint, {'?'});
574 } else {
575 for (int i = 0; i < r.count; ++i)
576 this->buf_[i] = result_value{static_cast<char>(r.bytes[i])};
577 len_ = r.count;
578 }
579 pos_ = 0;
580 }
581}
