63template <codec C, std::ranges::random_access_range R, transcode_error_kind E = transcode_error_kind::replacement>
64 requires legacy_byte_range<R> && detail::random_access_decode_codec<C>
65class random_access_whatwg_decode_view
66 : public std::ranges::view_interface<random_access_whatwg_decode_view<C, R, E>> {
67 R base_;
68
69 class iterator {
70 using base_iter = detail::compatible_iterator_t<R>;
71
72 base_iter current_{};
73
74 public:
75 using iterator_concept = std::random_access_iterator_tag;
76 using iterator_category = std::random_access_iterator_tag;
77 using value_type =
78 std::conditional_t<E == transcode_error_kind::expected, std::expected<char32_t, whatwg_error>, char32_t>;
79
80 // A failure, as this error kind reports it. In replacement mode the
81 // table lookup has already produced U+FFFD, so both arms agree.
82 static constexpr value_type error_result(whatwg_error e) {
83 if constexpr (E == transcode_error_kind::expected)
84 return value_type(std::unexpect, e);
85 else
86 return U'\xFFFD';
87 }
88 using difference_type = std::ptrdiff_t;
89 using reference = value_type;
90
91 constexpr iterator() = default;
92 constexpr explicit iterator(base_iter current) : current_(current) {}
93
94 constexpr const base_iter& base() const noexcept { return current_; }
95
96 constexpr value_type operator*() const {
97 const auto byte = static_cast<unsigned char>(*current_);
98 if constexpr (C == codec::x_user_defined) {
99 return detail::decode_x_user_defined_code_unit(byte);
100 } else {
101 const auto cp = detail::decode_random_access_code_unit(byte, detail::random_access_decode_table<C>());
102 if (cp == U'\xFFFD' && byte >= 0x80 && detail::random_access_decode_table<C>()[byte - 0x80] == 0)
103 return error_result(whatwg_error::invalid_byte);
104 return cp;
105 }
106 }
107
108 constexpr value_type operator[](difference_type n) const { return *(*this + n); }
109
110 constexpr iterator& operator++() {
111 ++current_;
112 return *this;
113 }
114
115 constexpr iterator operator++(int) {
116 auto tmp = *this;
117 ++*this;
118 return tmp;
119 }
120
121 constexpr iterator& operator--() {
122 --current_;
123 return *this;
124 }
125
126 constexpr iterator operator--(int) {
127 auto tmp = *this;
128 --*this;
129 return tmp;
130 }
131
132 constexpr iterator& operator+=(difference_type n) {
133 current_ += n;
134 return *this;
135 }
136
137 constexpr iterator& operator-=(difference_type n) {
138 current_ -= n;
139 return *this;
140 }
141
142 constexpr iterator operator+(difference_type n) const {
143 auto tmp = *this;
144 return tmp += n;
145 }
146
147 constexpr iterator operator-(difference_type n) const {
148 auto tmp = *this;
149 return tmp -= n;
150 }
151
152 constexpr difference_type operator-(const iterator& other) const { return current_ - other.current_; }
153
154 constexpr friend iterator operator+(difference_type n, iterator it) { return it += n; }
155
156 constexpr auto operator<=>(const iterator&) const = default;
157 };
158
159 public:
160 constexpr explicit random_access_whatwg_decode_view(R base) : base_(std::move(base)) {}
161
162 constexpr const R& base() const& noexcept { return base_; }
163 constexpr R base() && { return std::move(base_); }
164
165 constexpr iterator begin() { return iterator(std::ranges::begin(base_)); }
166
167 constexpr iterator begin() const
168 requires detail::const_iterator_compatible_range<R>
169 {
170 return iterator(std::ranges::begin(base_));
171 }
172
173 constexpr iterator end() { return iterator(std::ranges::end(base_)); }
174
175 constexpr iterator end() const
176 requires detail::const_iterator_compatible_range<R> && std::ranges::range<const R>
177 {
178 return iterator(std::ranges::end(base_));
179 }
180
181 constexpr auto size() const
182 requires std::ranges::sized_range<R>
183 {
184 return std::ranges::size(base_);
185 }
186};

192template <codec C, std::ranges::input_range R, transcode_error_kind E = transcode_error_kind::replacement>
193 requires legacy_byte_range<R>
194class whatwg_decode_view : public std::ranges::view_interface<whatwg_decode_view<C, R, E>> {
195 R base_;
196
197 class iterator {
198 using base_iter = detail::compatible_iterator_t<R>;
199 using base_sent = detail::compatible_sentinel_t<R>;
200 using result_t =
201 std::conditional_t<E == transcode_error_kind::expected, std::expected<char32_t, whatwg_error>, char32_t>;
202
203 // Whether the loaded value is this code point, under either error kind.
204 constexpr bool holds(char32_t cp) const {
205 if constexpr (E == transcode_error_kind::expected)
206 return value_.has_value() && *value_ == cp;
207 else
208 return value_ == cp;
209 }
210
211 // A failure, as this error kind reports it.
212 static constexpr result_t error_result(whatwg_error e) {
213 if constexpr (E == transcode_error_kind::expected)
214 return result_t(std::unexpect, e);
215 else
216 return U'\xFFFD';
217 }
218
219 base_iter current_{};
220 base_sent end_{};
221 result_t value_{};
222 bool done_{false};
223 int pending_count_{0};
224 unsigned char pending_[2]{};
225 char32_t pending_cp_{};
226 bool has_pending_cp_{false};
227 int iso2022jp_state_{0};
228 int iso2022jp_output_state_{0};
229 bool iso2022jp_output_flag_{false};
230 unsigned char iso2022jp_lead_{0};
231 unsigned char gb_replay_[3]{};
232 int gb_replay_count_{0};
233 int gb_replay_pos_{0};
234
235 constexpr void load();
236
237 public:
238 constexpr iterator() = default;
239
240 static constexpr iterator terminal()
241 requires std::ranges::forward_range<R>
242 {
243 iterator it;
244 it.done_ = true;
245 return it;
246 }
247
248 using iterator_concept =
249 std::conditional_t<std::ranges::forward_range<R>, std::forward_iterator_tag, std::input_iterator_tag>;
250 using iterator_category = iterator_concept;
251 using value_type = result_t;
252 using difference_type = std::ptrdiff_t;
253 using reference = result_t;
254
255 constexpr iterator(base_iter current, base_sent end);
256
257 constexpr const base_iter& base() const noexcept { return current_; }
258
259 constexpr result_t operator*() const;
260 constexpr iterator& operator++();
261 constexpr iterator operator++(int)
262 requires std::ranges::forward_range<R>;
263 constexpr void operator++(int)
264 requires(!std::ranges::forward_range<R>);
265
266 constexpr friend bool operator==(const iterator& lhs, const iterator& rhs)
267 requires std::ranges::forward_range<R>
268 {
269 if (lhs.done_ || rhs.done_)
270 return lhs.done_ == rhs.done_;
271 return lhs.current_ == rhs.current_ && lhs.value_ == rhs.value_ && lhs.done_ == rhs.done_ &&
272 lhs.pending_count_ == rhs.pending_count_ && lhs.pending_cp_ == rhs.pending_cp_ &&
273 lhs.has_pending_cp_ == rhs.has_pending_cp_ && lhs.iso2022jp_state_ == rhs.iso2022jp_state_ &&
274 lhs.iso2022jp_output_state_ == rhs.iso2022jp_output_state_ &&
275 lhs.iso2022jp_output_flag_ == rhs.iso2022jp_output_flag_ &&
276 lhs.iso2022jp_lead_ == rhs.iso2022jp_lead_ && lhs.gb_replay_count_ == rhs.gb_replay_count_ &&
277 lhs.gb_replay_pos_ == rhs.gb_replay_pos_ &&
278 std::equal(std::begin(lhs.pending_), std::end(lhs.pending_), std::begin(rhs.pending_)) &&
279 std::equal(std::begin(lhs.gb_replay_), std::end(lhs.gb_replay_), std::begin(rhs.gb_replay_));
280 }
281
282 constexpr friend bool operator==(const iterator& it, std::default_sentinel_t) { return it.done_; }
283 };
284
285 public:
286 constexpr explicit whatwg_decode_view(R base);
287
288 constexpr const R& base() const& noexcept { return base_; }
289 constexpr R base() && { return std::move(base_); }
290
291 constexpr iterator begin();
292 constexpr iterator begin() const
293 requires detail::const_iterator_compatible_range<R> && detail::const_sentinel_compatible_range<R>;
294 constexpr iterator end()
295 requires std::ranges::forward_range<R> && std::ranges::common_range<R>;
296 constexpr iterator end() const
297 requires std::ranges::forward_range<const R> && std::ranges::common_range<const R> &&
298 detail::const_iterator_compatible_range<R> && detail::const_sentinel_compatible_range<R>;
299 constexpr std::default_sentinel_t end() const;
300};

302template <codec C, transcode_error_kind E = transcode_error_kind::replacement>
303struct whatwg_decode_closure {
304 template <legacy_byte_range R>
305 constexpr auto operator()(R&& r) const;
306
307 template <legacy_byte_range R>
308 constexpr friend auto operator|(R&& r, const whatwg_decode_closure& self) {
309 return self(std::forward<R>(r));
310 }
311
312 template <typename R>
313 requires std::is_array_v<std::remove_cvref_t<R>>
314 friend auto operator|(R&&, const whatwg_decode_closure&) {
315 static_assert(!std::is_array_v<std::remove_cvref_t<R>>,
316 "transcode: raw arrays are not valid input to whatwg_decode; "
317 "use beman::transcoding::views::null_term to get a range from a "
318 "null-terminated string");
319 }
320};

407template <codec C, std::ranges::input_range R, transcode_error_kind E>
408 requires legacy_byte_range<R>
409constexpr void whatwg_decode_view<C, R, E>::iterator::load() {
410 if (has_pending_cp_) {
411 value_ = pending_cp_;
412 has_pending_cp_ = false;
413 return;
414 }
415 if (current_ == end_) {
416 if (pending_count_ == 0 && iso2022jp_state_ <= 3 && gb_replay_pos_ >= gb_replay_count_) {
417 done_ = true;
418 return;
419 }
420 }
421 if constexpr (C == codec::replacement) {
422 while (current_ != end_)
423 ++current_;
424 value_ = error_result(whatwg_error::invalid_byte);
425 } else if constexpr (C == codec::utf_8) {
426 auto r = detail::utf8_decode_one(current_, end_);
427 if (r.is_error)
428 value_ = error_result(r.error);
429 else
430 value_ = r.code_point;
431 } else if constexpr (C == codec::x_user_defined) {
432 value_ = detail::x_user_defined_decode_one(current_, end_);
433 } else if constexpr (C == codec::windows_1252) {
434 auto r = detail::single_byte_decode_one(current_, end_, detail::tables::windows_1252);
435 if (r.is_error)
436 std::unreachable();
437 else
438 value_ = r.code_point;
439 } else if constexpr (detail::single_byte_table_codec<C>) {
440 auto r = detail::single_byte_decode_one(current_, end_, detail::single_byte_decode_table<C>());
441 if (r.is_error)
442 value_ = error_result(r.error);
443 else
444 value_ = r.code_point;
445 } else if constexpr (C == codec::utf_16be || C == codec::utf_16le) {
446 unsigned char b0;
447 unsigned char b1;
448 if (pending_count_ > 0) {
449 b0 = pending_[0];
450 b1 = pending_[1];
451 pending_count_ = 0;
452 } else {
453 b0 = static_cast<unsigned char>(*current_);
454 ++current_;
455 if (current_ == end_) {
456 value_ = error_result(whatwg_error::truncated_sequence);
457 return;
458 }
459 b1 = static_cast<unsigned char>(*current_);
460 ++current_;
461 }
462 char16_t unit;
463 if constexpr (C == codec::utf_16be)
464 unit = static_cast<char16_t>((static_cast<unsigned>(b0) << 8) | b1);
465 else
466 unit = static_cast<char16_t>((static_cast<unsigned>(b1) << 8) | b0);
467
468 if (unit >= 0xD800 && unit <= 0xDBFF) {
469 if (current_ == end_) {
470 value_ = error_result(whatwg_error::truncated_sequence);
471 return;
472 }
473 auto b2 = static_cast<unsigned char>(*current_);
474 ++current_;
475 if (current_ == end_) {
476 value_ = error_result(whatwg_error::truncated_sequence);
477 return;
478 }
479 auto b3 = static_cast<unsigned char>(*current_);
480 ++current_;
481 char16_t low;
482 if constexpr (C == codec::utf_16be)
483 low = static_cast<char16_t>((static_cast<unsigned>(b2) << 8) | b3);
484 else
485 low = static_cast<char16_t>((static_cast<unsigned>(b3) << 8) | b2);
486 if (low >= 0xDC00 && low <= 0xDFFF) {
487 value_ = 0x10000 + ((static_cast<char32_t>(unit - 0xD800) << 10) | (low - 0xDC00));
488 } else {
489 value_ = error_result(whatwg_error::surrogate_code_point);
490 pending_[0] = b2;
491 pending_[1] = b3;
492 pending_count_ = 2;
493 }
494 } else if (unit >= 0xDC00 && unit <= 0xDFFF) {
495 value_ = error_result(whatwg_error::surrogate_code_point);
496 } else {
497 value_ = static_cast<char32_t>(unit);
498 }
499 } else if constexpr (C == codec::gbk || C == codec::gb18030) {
500 if (gb_replay_pos_ < gb_replay_count_) {
501 auto byte = gb_replay_[gb_replay_pos_++];
502 if (gb_replay_pos_ == gb_replay_count_) {
503 gb_replay_count_ = 0;
504 gb_replay_pos_ = 0;
505 }
506 if (byte < 0x80) {
507 value_ = static_cast<char32_t>(byte);
508 } else {
509 unsigned char buf[4];
510 buf[0] = byte;
511 int count = 1;
512 while (count < 4 && current_ != end_) {
513 buf[count++] = static_cast<unsigned char>(*current_);
514 ++current_;
515 }
516 const unsigned char* bp = buf;
517 const unsigned char* be = buf + count;
518 auto r = detail::gb18030_decode_one(bp, be);
519 int left = static_cast<int>(be - bp);
520 if (left > 0) {
521 gb_replay_count_ = left;
522 gb_replay_pos_ = 0;
523 for (int i = 0; i < left; ++i)
524 gb_replay_[i] = bp[i];
525 }
526 if (r.is_error)
527 value_ = error_result(r.error);
528 else
529 value_ = r.code_point;
530 }
531 return;
532 }
533 auto r = detail::gb18030_decode_one(current_, end_);
534 if (r.is_error) {
535 value_ = error_result(r.error);
536 if (r.replay_count > 0) {
537 gb_replay_count_ = r.replay_count;
538 gb_replay_pos_ = 0;
539 for (int i = 0; i < r.replay_count; ++i)
540 gb_replay_[i] = r.replay[i];
541 }
542 } else {
543 value_ = r.code_point;
544 }
545 } else if constexpr (C == codec::big5) {
546 auto r = detail::big5_decode_one(current_, end_);
547 if (r.is_error)
548 value_ = error_result(r.error);
549 else
550 value_ = r.code_point;
551 if (r.code_point2 != 0) {
552 pending_cp_ = r.code_point2;
553 has_pending_cp_ = true;
554 }
555 } else if constexpr (C == codec::shift_jis) {
556 auto r = detail::shift_jis_decode_one(current_, end_);
557 if (r.is_error)
558 value_ = error_result(r.error);
559 else
560 value_ = r.code_point;
561 } else if constexpr (C == codec::euc_jp) {
562 auto r = detail::euc_jp_decode_one(current_, end_);
563 if (r.is_error)
564 value_ = error_result(r.error);
565 else
566 value_ = r.code_point;
567 } else if constexpr (C == codec::iso_2022_jp) {
568 // States: 0=ASCII, 1=Roman, 2=Katakana, 3=Lead_Byte, 4=Trail_Byte,
569 // 5=Escape_Start, 6=Escape
570 while (true) {
571 unsigned char byte;
572 if (pending_count_ > 0) {
573 byte = pending_[0];
574 if (pending_count_ > 1)
575 pending_[0] = pending_[1];
576 --pending_count_;
577 } else if (current_ != end_) {
578 byte = static_cast<unsigned char>(*current_++);
579 } else {
580 switch (iso2022jp_state_) {
581 default: // 0=ASCII, 1=Roman, 2=Katakana, 3=Lead_Byte
582 done_ = true;
583 return;
584 case 4:
585 iso2022jp_state_ = iso2022jp_output_state_;
586 iso2022jp_output_flag_ = false;
587 value_ = error_result(whatwg_error::truncated_sequence);
588 return;
589 case 5:
590 iso2022jp_state_ = iso2022jp_output_state_;
591 iso2022jp_output_flag_ = false;
592 value_ = error_result(whatwg_error::truncated_sequence);
593 return;
594 case 6:
595 pending_[0] = iso2022jp_lead_;
596 pending_count_ = 1;
597 iso2022jp_lead_ = 0;
598 iso2022jp_state_ = iso2022jp_output_state_;
599 iso2022jp_output_flag_ = false;
600 value_ = error_result(whatwg_error::truncated_sequence);
601 return;
602 }
603 }
604
605 switch (iso2022jp_state_) {
606 case 5:
607 if (byte == 0x24 || byte == 0x28) {
608 iso2022jp_lead_ = byte;
609 iso2022jp_state_ = 6;
610 continue;
611 }
612 pending_[0] = byte;
613 pending_count_ = 1;
614 iso2022jp_state_ = iso2022jp_output_state_;
615 iso2022jp_output_flag_ = false;
616 value_ = error_result(whatwg_error::invalid_byte);
617 return;
618
619 case 6: {
620 auto lead = iso2022jp_lead_;
621 iso2022jp_lead_ = 0;
622 int new_state = -1;
623 if (lead == 0x28) {
624 if (byte == 0x42)
625 new_state = 0;
626 else if (byte == 0x4A)
627 new_state = 1;
628 else if (byte == 0x49)
629 new_state = 2;

632 new_state = 3;
633 }
634 if (new_state < 0) {
635 pending_[0] = lead;
636 pending_[1] = byte;
637 pending_count_ = 2;
638 iso2022jp_state_ = iso2022jp_output_state_;
639 iso2022jp_output_flag_ = false;
640 value_ = error_result(whatwg_error::invalid_byte);
641 return;
642 }
643 iso2022jp_output_state_ = new_state;
644 iso2022jp_state_ = new_state;
645 if (iso2022jp_output_flag_) {
646 iso2022jp_output_flag_ = false;
647 value_ = error_result(whatwg_error::invalid_byte);
648 return;
649 }
650 iso2022jp_output_flag_ = true;
651 continue;
652 }
653
654 case 3:
655 if (byte == 0x1B) {
656 iso2022jp_state_ = 5;
657 continue;
658 }
659 if (byte >= 0x21 && byte <= 0x7E) {
660 iso2022jp_lead_ = byte;
661 iso2022jp_state_ = 4;
662 continue;
663 }
664 iso2022jp_state_ = iso2022jp_output_state_;
665 iso2022jp_output_flag_ = false;
666 value_ = error_result(whatwg_error::invalid_byte);
667 return;
668
669 case 4: {
670 if (byte == 0x1B) {
671 iso2022jp_state_ = 5;
672 continue;
673 }
674 iso2022jp_state_ = iso2022jp_output_state_;
675 iso2022jp_output_flag_ = false;
676 if (byte >= 0x21 && byte <= 0x7E) {
677 int pointer = ((static_cast<int>(iso2022jp_lead_) - 0x21) * 94) + (static_cast<int>(byte) - 0x21);
678 auto cp = detail::tables::shift_jis[pointer];
679 if (cp != 0) {
680 value_ = cp;
681 return;
682 }
683 }
684 value_ = error_result(whatwg_error::invalid_byte);
685 return;
686 }
687
688 default: // 0=ASCII, 1=Roman, 2=Katakana
689 if (byte == 0x1B) {
690 iso2022jp_state_ = 5;
691 continue;
692 }
693 if (byte == 0x0E || byte == 0x0F) {
694 iso2022jp_output_flag_ = false;
695 value_ = error_result(whatwg_error::invalid_byte);
696 return;
697 }
698 iso2022jp_output_flag_ = false;
699 if (iso2022jp_state_ == 0) {
700 if (byte <= 0x7F)
701 value_ = static_cast<char32_t>(byte);
702 else
703 value_ = error_result(whatwg_error::invalid_byte);
704 return;
705 }
706 if (iso2022jp_state_ == 1) {
707 if (byte == 0x5C) {
708 value_ = U'\x00A5';
709 return;
710 }
711 if (byte == 0x7E) {
712 value_ = U'\x203E';
713 return;
714 }
715 if (byte <= 0x7F)
716 value_ = static_cast<char32_t>(byte);
717 else
718 value_ = error_result(whatwg_error::invalid_byte);
719 return;
720 }
721 // Katakana: 0x21-0x5F → U+FF61-U+FF9F
722 if (byte >= 0x21 && byte <= 0x5F)
723 value_ = static_cast<char32_t>(0xFF61 + byte - 0x21);
724 else
725 value_ = error_result(whatwg_error::invalid_byte);
726 return;
727 }
728 }
729 } else if constexpr (C == codec::euc_kr) {
730 auto r = detail::euc_kr_decode_one(current_, end_);
731 if (r.is_error)
732 value_ = error_result(r.error);
733 else
734 value_ = r.code_point;
735 }
736}
