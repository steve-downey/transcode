3// A parser-combinator kit for the MIME reader example, copied from specgen
4// (include/beman/specgen/parse/parser.hpp, the same author and license), which
5// adapted it from compile-time-scheme (src/smd/smdscheme/parser/parser.hpp).
6//
7// What is kept is the discipline: a parser is any callable from a cursor to
8// `std::expected<parse_state<T>, parse_error>`, and the ordered choice
9// `operator|` tries its second alternative only when the first failed without
10// consuming input. A branch that has eaten input has committed, so a failure
11// part-way through a token is reported where it happened and not papered over
12// by a sibling. `attempt` opts one parser out of that rule.
13//
14// What changed: the namespace is example-local, the input is bytes, and the
15// combinators the MIME grammar does not use (`digits`, `lexeme`, ...) are
16// dropped. New are `lift` (n-ary `map`, so a production reads left to right
17// like its ABNF), `keyword_ci`, `take_while` and `take_while1` (a run of bytes
18// as a view of the input), `attempt`, `followed_by`, `eof` and `scan_until`.
19

39// A failure: the offset in the input and a static description of what was
40// expected there. The offset is also the commit signal for `operator|`.
41struct parse_error {
42 std::size_t offset = 0;
43 const char* message = "";
44};

87// A parser with its type erased. Every combinator nests its operands by
88// value, so the type of a production built from others would contain the whole
89// grammar beneath it, and a large grammar takes gigabytes to compile. A
90// grammar's named productions are `rule<T>`s, which cuts the nesting at each
91// of them. The price is one indirect call per rule invocation.
92template <class T>
93class rule {
94 public:
95 rule() = default;
96
97 template <parser_like P>
98 requires(!std::same_as<std::remove_cvref_t<P>, rule>)
99 rule(P p);
100
101 parse_result<T> operator()(cursor cur) const;
102
103 private:
104 std::function<parse_result<T>(cursor)> f_;
105};
106
107template <class T>
108template <parser_like P>
109 requires(!std::same_as<std::remove_cvref_t<P>, rule<T>>)
110rule<T>::rule(P p) : f_{std::move(p)} {}
111
112template <class T>
113parse_result<T> rule<T>::operator()(cursor cur) const {
114 return f_(cur);
115}

117// Succeeds on one byte satisfying `pred`.
118[[nodiscard]] constexpr auto satisfy(auto pred, const char* expected) {
119 return parser{[pred, expected](cursor cur) -> parse_result<char> {
120 if (!cur.empty() && pred(cur.peek())) {
121 return parse_state<char>{cur.peek(), cur.bump()};
122 }
123 return std::unexpected(parse_error{cur.offset(), expected});
124 }};
125}
126
127[[nodiscard]] constexpr auto char_p(char expected) {
128 return satisfy([expected](char c) { return c == expected; }, "expected a particular byte");
129}

131// A run of zero or more bytes satisfying `pred`, as a view of the input.
132[[nodiscard]] constexpr auto take_while(auto pred) {
133 return parser{[pred](cursor cur) -> parse_result<std::string_view> {
134 std::string_view rest = cur.remaining();
135 std::size_t n = 0;
136 while (n < rest.size() && pred(rest[n])) {
137 ++n;
138 }
139 return parse_state<std::string_view>{rest.substr(0, n), cur.advance(n)};
140 }};
141}
142
143// A run of one or more bytes satisfying `pred`.
144[[nodiscard]] constexpr auto take_while1(auto pred, const char* expected) {
145 return parser{[pred, expected](cursor cur) -> parse_result<std::string_view> {
146 std::string_view rest = cur.remaining();
147 std::size_t n = 0;
148 while (n < rest.size() && pred(rest[n])) {
149 ++n;
150 }
151 if (n == 0) {
152 return std::unexpected(parse_error{cur.offset(), expected});
153 }
154 return parse_state<std::string_view>{rest.substr(0, n), cur.advance(n)};
155 }};
156}

158// The literal `text`, byte for byte. A mismatch on the first byte fails
159// without consuming, so `operator|` can still try a sibling; a mismatch after
160// that has committed.
161[[nodiscard]] constexpr auto keyword(std::string text) {
162 return parser{[text = std::move(text)](cursor cur) -> parse_result<std::string_view> {
163 std::string_view rest = cur.remaining();
164 for (std::size_t i = 0; i < text.size(); ++i) {
165 if (i == rest.size() || rest[i] != text[i]) {
166 return std::unexpected(parse_error{cur.offset() + i, "expected a literal"});
167 }
168 }
169 return parse_state<std::string_view>{rest.substr(0, text.size()), cur.advance(text.size())};
170 }};
171}

173constexpr char ascii_lower(char c) { return (c >= 'A' && c <= 'Z') ? static_cast<char>(c - 'A' + 'a') : c; }
174
175// ABNF strings are case-insensitive: `keyword_ci("base64")` matches "BASE64".
176// `text` is lower case.
177[[nodiscard]] constexpr auto keyword_ci(std::string text) {
178 return parser{[text = std::move(text)](cursor cur) -> parse_result<std::string_view> {
179 std::string_view rest = cur.remaining();
180 for (std::size_t i = 0; i < text.size(); ++i) {
181 if (i == rest.size() || ascii_lower(rest[i]) != text[i]) {
182 return std::unexpected(parse_error{cur.offset() + i, "expected a literal"});
183 }
184 }
185 return parse_state<std::string_view>{rest.substr(0, text.size()), cur.advance(text.size())};
186 }};
187}

189// Succeeds only at the end of the input.
190[[nodiscard]] constexpr auto eof() {
191 return parser{[](cursor cur) -> parse_result<unit> {
192 if (!cur.empty()) {
193 return std::unexpected(parse_error{cur.offset(), "expected the end of the input"});
194 }
195 return parse_state<unit>{unit{}, cur};
196 }};
197}

199// Runs each parser in order and applies `f` to their values. The first
200// failure is the result. `lift(f, p)` is `map`.
201template <class F, parser_like... Ps>
202[[nodiscard]] constexpr auto lift(F f, Ps... ps) {
203 using V = std::invoke_result_t<const F&, value_of<Ps>...>;
204 return parser{[f, ps = std::tuple{ps...}](cursor cur) -> parse_result<V> {
205 std::tuple<std::optional<value_of<Ps>>...> values;
206 parse_error error{};
207 auto step = [&]<std::size_t I>() {
208 auto r = std::get<I>(ps)(cur);
209 if (!r) {
210 error = r.error();
211 return false;
212 }
213 std::get<I>(values) = std::move(r->value);
214 cur = r->rest;
215 return true;
216 };
217 auto run = [&]<std::size_t... Is>(std::index_sequence<Is...>) {
218 return (step.template operator()<Is>() && ...);
219 };
220 if (!run(std::index_sequence_for<Ps...>{})) {
221 return std::unexpected(error);
222 }
223 return std::apply([&](auto&... v) { return parse_state<V>{f(std::move(*v)...), cur}; }, values);
224 }};
225}

227template <parser_like P>
228[[nodiscard]] constexpr auto map(P p, auto f) {
229 return lift(std::move(f), std::move(p));
230}

232// `pa pb`, keeping the value of `pa`.
233template <parser_like PA, parser_like PB>
234[[nodiscard]] constexpr auto sequence_left(PA pa, PB pb) {
235 return lift([](auto a, auto&&) { return a; }, std::move(pa), std::move(pb));
236}

238// `pa pb`, keeping the value of `pb`.
239template <parser_like PA, parser_like PB>
240[[nodiscard]] constexpr auto sequence_right(PA pa, PB pb) {
241 return lift([](auto&&, auto b) { return b; }, std::move(pa), std::move(pb));
242}

244// Ordered choice, `pa / pb`: `pb` is tried only when `pa` failed at the offset
245// where it started. Both must produce the same type.
246template <parser_like PA, parser_like PB>
247[[nodiscard]] constexpr auto operator|(PA pa, PB pb) {
248 return parser{[pa, pb](cursor cur) {
249 auto ra = pa(cur);
250 if (ra.has_value() || ra.error().offset != cur.offset()) {
251 return ra;
252 }
253 return pb(cur);
254 }};
255}

257// `p` with its failure moved back to where it started, so that `operator|`
258// will try the next alternative even if `p` consumed input before failing.
259template <parser_like P>
260[[nodiscard]] constexpr auto attempt(P p) {
261 return parser{[p](cursor cur) {
262 auto r = p(cur);
263 if (!r.has_value()) {
264 r.error().offset = cur.offset();
265 }
266 return r;
267 }};
268}

270// Succeeds, consuming nothing, when `p` would succeed here: `&p`.
271template <parser_like P>
272[[nodiscard]] constexpr auto followed_by(P p) {
273 return parser{[p](cursor cur) -> parse_result<unit> {
274 auto r = p(cur);
275 if (!r.has_value()) {
276 return std::unexpected(parse_error{cur.offset(), r.error().message});
277 }
278 return parse_state<unit>{unit{}, cur};
279 }};
280}

282// `*p`: zero or more, collected. Always succeeds, and stops at the first
283// failure of `p`, committed or not. `p` must consume input when it succeeds.
284template <parser_like P>
285[[nodiscard]] constexpr auto many(P p) {
286 return parser{[p](cursor cur) {
287 using V = value_of<P>;
288 std::vector<V> result;
289 while (auto r = p(cur)) {
290 result.push_back(std::move(r->value));
291 cur = r->rest;
292 }
293 return parse_result<std::vector<V>>{parse_state<std::vector<V>>{std::move(result), cur}};
294 }};
295}
296
297// `1*p`: one or more, collected.
298template <parser_like P>
299[[nodiscard]] constexpr auto some(P p) {
300 return parser{[p](cursor cur) {
301 using V = value_of<P>;
302 std::vector<V> result;
303 auto first = p(cur);
304 if (!first.has_value()) {
305 return parse_result<std::vector<V>>{std::unexpected(first.error())};
306 }
307 result.push_back(std::move(first->value));
308 cur = first->rest;
309 while (auto r = p(cur)) {
310 result.push_back(std::move(r->value));
311 cur = r->rest;
312 }
313 return parse_result<std::vector<V>>{parse_state<std::vector<V>>{std::move(result), cur}};
314 }};
315}

317// `[p]`: `p` if it matches without having failed part-way, else nothing.
318template <parser_like P>
319[[nodiscard]] constexpr auto opt(P p) {
320 return parser{[p](cursor cur) {
321 using V = value_of<P>;
322 auto r = p(cur);
323 if (r.has_value()) {
324 return parse_result<std::optional<V>>{parse_state<std::optional<V>>{std::move(r->value), r->rest}};
325 }
326 if (r.error().offset != cur.offset()) {
327 return parse_result<std::optional<V>>{std::unexpected(r.error())};
328 }
329 return parse_result<std::optional<V>>{parse_state<std::optional<V>>{std::nullopt, cur}};
330 }};
331}

333template <class T>
334struct scanned {
335 std::string_view skipped; // the bytes passed over
336 T value; // what `p` produced where it matched
337};

339// Skips bytes until `p` matches, and yields the skipped bytes with `p`'s value.
340// A failure of `p` at one offset only moves the scan on to the next, which is
341// what a preamble, a body part, or an epilogue is: everything up to a marker.
342template <parser_like P>
343[[nodiscard]] constexpr auto scan_until(P p, const char* expected) {
344 return parser{[p, expected](cursor cur) -> parse_result<scanned<value_of<P>>> {
345 for (std::size_t n = 0; n <= cur.remaining().size(); ++n) {
346 if (auto r = p(cur.advance(n))) {
347 return parse_state<scanned<value_of<P>>>{
348 scanned<value_of<P>>{cur.remaining().substr(0, n), std::move(r->value)}, r->rest};
349 }
350 }
351 return std::unexpected(parse_error{cur.offset(), expected});
352 }};
353}
