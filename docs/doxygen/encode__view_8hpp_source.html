440template <text_encoder Codec, std::ranges::input_range R, error_mode E>
441 requires unicode_scalar_range<R>
442constexpr void encode_view<Codec, R, E>::iterator::load() {
443 if (current_ == end_) {
444 if constexpr (flushable_text_encoder<Codec>) {
445 auto flushed = codec_.flush();
446 if (flushed && (flushed->is_error || flushed->count > 0)) {
447 // The same rule as an encode_one error: bytes supplied with an
448 // error are the replacement in replacement mode.
449 if (flushed->is_error && (E == error_mode::report || flushed->count == 0)) {
450 is_error_ = true;
451 error_ = whatwg::error::unmapped_codepoint;
452 buf_[0] = '?';
453 len_ = 1;
454 } else {
455 is_error_ = false;
456 for (int i = 0; i < flushed->count; ++i)
457 buf_[static_cast<std::size_t>(i)] = flushed->bytes[static_cast<std::size_t>(i)];
458 len_ = flushed->count;
459 }
460 pos_ = 0;
461 return;
462 }
463 }
464 done_ = true;
465 return;
466 }
467 const auto scalar = static_cast<char32_t>(*current_);
468 ++current_;
469 const auto validation = detail::validate_utf32(scalar);
470 if constexpr (E == error_mode::report) {
471 if (validation.is_error) {
472 is_error_ = true;
473 error_ = validation.error;
474 buf_[0] = '?';
475 len_ = 1;
476 pos_ = 0;
477 return;
478 }
479 }
480 auto r = codec_.encode_one(validation.code_point);
481 // An error that carries bytes, in replacement mode, is the codec's own
482 // replacement and is yielded as it is; otherwise the view substitutes '?'.
483 if (r.is_error && (E == error_mode::report || r.count == 0)) {
484 is_error_ = true;
485 error_ = whatwg::error::unmapped_codepoint;
486 buf_[0] = '?';
487 len_ = 1;
488 } else {
489 is_error_ = false;
490 // Byte by byte under `count`, unrolled: a loop bounded by the runtime
491 // `count` becomes a call to memcpy, and a whole-array copy makes the
492 // compiler build every byte of `r`, so either spills it to the stack.
493 static_assert(std::extent_v<decltype(buf_)> == std::extent_v<decltype(r.bytes)>);
494 [&]<std::size_t... I>(std::index_sequence<I...>) {
495 (void)((static_cast<int>(I) < r.count && (buf_[I] = r.bytes[I], true)) && ...);
496 }(std::make_index_sequence<std::extent_v<decltype(buf_)>>{});
497 len_ = r.count;
498 }
499 pos_ = 0;
500}

502template <text_encoder Codec, std::ranges::input_range R, error_mode E>
503 requires unicode_scalar_range<R>
506constexpr auto encode_view<Codec, R, E>::iterator::operator*() const -> result_t {
507 // load() stores the '?' that replaces a failure in buf_, so only the
508 // report view tests is_error_ here; it reports why.
509 if constexpr (E == error_mode::report) {
510 if (is_error_)
511 return result_t(std::unexpect, error_);
512 }
513 return static_cast<char>(buf_[static_cast<std::size_t>(pos_)]);
514}
