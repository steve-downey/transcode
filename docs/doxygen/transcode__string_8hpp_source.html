56template <whatwg::encoding C>
57std::u32string transcode_decode_all(std::span<const char> src) {
58 std::u32string result;
59 for (char32_t cp : src | decode<C>)
60 result.push_back(cp);
61 return result;
62}

64template <whatwg::encoding C>
65std::string transcode_encode_all(std::u32string_view src) {
66 std::string result;
67 for (char b : src | encode<C>)
68 result.push_back(b);
69 return result;
70}

95inline std::optional<std::string>
96transcode_string(std::span<const char> src, whatwg::encoding from, whatwg::encoding to) {
97 if (!has_encoder(to))
98 return std::nullopt;
99
100 // Each encoding is examined once, and the loop it selects is the one a
101 // caller naming the encoding at compile time would get.
102 const std::u32string intermediate = beman::transcoding::detail::with_encoding(
103 from, [&]<whatwg::encoding E>(std::integral_constant<whatwg::encoding, E>) {
104 return detail::transcode_decode_all<E>(src);
105 });
106 return beman::transcoding::detail::with_encoding(
107 to, [&]<whatwg::encoding E>(std::integral_constant<whatwg::encoding, E>) -> std::optional<std::string> {
108 if constexpr (beman::transcoding::detail::whatwg_encode_codec<E>)
109 return detail::transcode_encode_all<E>(intermediate);
110 else
111 return std::nullopt;
112 });
113}

121inline std::optional<std::string>
122transcode_string(std::span<const char> src, std::string_view from_label, std::string_view to_label) {
123 auto from = whatwg::get_encoding(from_label);
124 auto to = whatwg::get_encoding(to_label);
125 if (!from || !to)
126 return std::nullopt;
127 return transcode_string(src, *from, *to);
128}
