79template <encoding C>
80template <std::input_iterator I, std::sentinel_for<I> S>
81constexpr decode_result decoder<C>::decode_one(I& current, S end) {
82 const auto failure = [](whatwg::error e) { return decode_result{U'\xFFFD', e, true}; };
83 const auto success = [](char32_t cp) { return decode_result{cp, {}, false}; };
84 if constexpr (C == encoding::big5) {
85 if (state_.has_pending) {
86 state_.has_pending = false;
87 return success(state_.code_point);
88 }
89 }
90 if constexpr (C == encoding::replacement) {
91 while (current != end)
92 ++current;
93 return failure(whatwg::error::invalid_byte);
94 } else if constexpr (C == encoding::utf_8) {
95 auto r = detail::utf8_decode_one(current, end);
96 return r.is_error ? failure(r.error) : success(r.code_point);
97 } else if constexpr (C == encoding::x_user_defined) {
98 return success(detail::x_user_defined_decode_one(current, end));
99 } else if constexpr (C == encoding::windows_1252) {
100 auto r = detail::single_byte_decode_one(current, end, detail::tables::windows_1252);
101 if (r.is_error)
102 std::unreachable();
103 return success(r.code_point);
104 } else if constexpr (detail::single_byte_table_codec<C>) {
105 auto r = detail::single_byte_decode_one(current, end, detail::single_byte_decode_table<C>());
106 return r.is_error ? failure(r.error) : success(r.code_point);
107 } else if constexpr (C == encoding::utf_16be || C == encoding::utf_16le) {
108 auto r = detail::utf16_decode_one<C == encoding::utf_16be>(state_, current, end);
109 return r.is_error ? failure(r.error) : success(r.code_point);
110 } else if constexpr (C == encoding::gbk || C == encoding::gb18030) {
111 auto r = detail::gb18030_decode_one(state_, current, end);
112 return r.is_error ? failure(r.error) : success(r.code_point);
113 } else if constexpr (C == encoding::big5) {
114 auto r = detail::big5_decode_one(current, end);
115 if (r.code_point2 != 0) {
116 state_.code_point = r.code_point2;
117 state_.has_pending = true;
118 }
119 return r.is_error ? failure(r.error) : success(r.code_point);
120 } else if constexpr (C == encoding::shift_jis) {
121 auto r = detail::shift_jis_decode_one(current, end);
122 return r.is_error ? failure(r.error) : success(r.code_point);
123 } else if constexpr (C == encoding::euc_jp) {
124 auto r = detail::euc_jp_decode_one(current, end);
125 return r.is_error ? failure(r.error) : success(r.code_point);
126 } else if constexpr (C == encoding::iso_2022_jp) {
127 auto r = detail::iso2022jp_decode_one(state_, current, end);
128 if (r.done)
129 return decode_result{.is_empty = true};
130 return r.is_error ? failure(r.error) : success(r.code_point);
131 } else {
132 static_assert(C == encoding::euc_kr);
133 auto r = detail::euc_kr_decode_one(current, end);
134 return r.is_error ? failure(r.error) : success(r.code_point);
135 }
136}

143template <encoding C>
144template <std::input_iterator I, std::sentinel_for<I> S>
145constexpr decode_result decoder<C>::decode_first(I& current, S end)
146 requires detail::strips_leading_bom<C>
147{
148 auto r = decode_one(current, end);
149 if (!r.is_error && r.code_point == U'\xFEFF') {
150 if (current == end && state_.at_end())
151 return decode_result{.is_empty = true};
152 return decode_one(current, end);
153 }
154 return r;
155}

157template <encoding C>
160constexpr char32_t decoder<C>::decode_byte(unsigned char byte) const
161 requires detail::random_access_decode_codec<C>
162{
163 if constexpr (C == encoding::x_user_defined)
164 return detail::decode_x_user_defined_code_unit(byte);
165 else
166 return detail::decode_random_access_code_unit(byte, detail::random_access_decode_table<C>());
167}

169template <encoding C>
174constexpr std::optional<decode_result> decoder<C>::flush()
175 requires(!std::same_as<detail::decode_state_t<C>, detail::no_decode_state>)
176{
177 if (state_.at_end())
178 return std::nullopt;
179 const unsigned char* none = nullptr;
180 auto r = decode_one(none, none);
181 if (r.is_empty)
182 return std::nullopt;
183 return r;
184}
