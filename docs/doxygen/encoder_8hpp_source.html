65template <encoding C>
66 requires detail::whatwg_encode_codec<C>
73constexpr encode_result encoder<C>::encode_one(char32_t scalar) {
74 // Each detail encoder builds the encode_result itself, and it is returned
75 // as a prvalue: a copy here, or a whole-array copy anywhere on the way to
76 // the view's buffer, makes GCC spill the result to the stack per element.
77 if constexpr (C == encoding::utf_8) {
78 return detail::utf8_encode_one(scalar);
79 } else if constexpr (detail::random_access_encode_codec<C>) {
80 const auto r = detail::single_byte_encode_one<detail::random_access_encode_table<C>()>(scalar);
81 encode_result out;
82 out.bytes[0] = r.byte;
83 out.count = r.is_error ? 0 : 1;
84 out.is_error = r.is_error;
85 return out;
86 } else if constexpr (C == encoding::gbk) {
87 return detail::gbk_encode_one(scalar);
88 } else if constexpr (C == encoding::gb18030) {
89 return detail::gb18030_encode_one(scalar);
90 } else if constexpr (C == encoding::big5) {
91 return detail::big5_encode_one(scalar);
92 } else if constexpr (C == encoding::shift_jis) {
93 return detail::shift_jis_encode_one(scalar);
94 } else if constexpr (C == encoding::euc_jp) {
95 return detail::euc_jp_encode_one(scalar);
96 } else if constexpr (C == encoding::iso_2022_jp) {
97 return detail::iso2022jp_encode_one(state_, scalar);
98 } else {
99 static_assert(C == encoding::euc_kr);
100 return detail::euc_kr_encode_one(scalar);
101 }
102}

104template <encoding C>
105 requires detail::whatwg_encode_codec<C>
108constexpr std::optional<unsigned char> encoder<C>::encode_byte(char32_t scalar) const
109 requires detail::random_access_encode_codec<C>
110{
111 // Straight from the table into the optional: going through
112 // single_byte_encode_one's {byte, is_error} pair costs GCC a second test
113 // per element.
114 if (scalar < 0x80)
115 return static_cast<unsigned char>(scalar);
116 const int pointer = detail::reverse_index_for<detail::random_access_encode_table<C>()>.find(scalar);
117 if (pointer < 0)
118 return std::nullopt;
119 return static_cast<unsigned char>(0x80 + pointer);
120}

122template <encoding C>
123 requires detail::whatwg_encode_codec<C>
127constexpr std::optional<encode_result> encoder<C>::flush()
128 requires(C == encoding::iso_2022_jp)
129{
130 const auto r = detail::iso2022jp_encode_flush(state_);
131 if (r.count == 0)
132 return std::nullopt;
133 return r;
134}
