19// A message or a body part. `body` is a view of the message's bytes and is
20// still transfer-encoded. A multipart entity has its parts, parsed in turn,
21// and a leaf has none. A part with no Content-Type is text/plain, and one with
22// no Content-Transfer-Encoding is 7bit (RFC 2045 sections 5.2 and 6.1).
23struct entity {
24 std::vector<header_field> headers;
25 media_type type;
26 transfer_encoding encoding = transfer_encoding::seven_bit;
27 std::string_view body;
28 std::vector<entity> parts;
29
30 // The first header field named `name`, in any case.
31 const header_field* find(std::string_view name) const;
32
33 bool is_multipart() const;
34 bool is_text() const;
35
36 // "type/subtype"
37 std::string content_type() const;
38
39 // The charset parameter of a text part, or "us-ascii" when it has none
40 // (RFC 2045 section 5.2), in lower case; empty for a part that is not text.
41 std::string charset() const;
42};

51// The text of a text leaf's body: its transfer encoding undone and its
