12// iconv_error — error categories that map to POSIX iconv errno values.
13// Used only by iconv_transcode_or_error_view and iconv_transcode_to_or_error.
14// Kept separate from whatwg::error because iconv reports at the OS level:
15// EILSEQ (invalid_sequence), EINVAL (incomplete_sequence), E2BIG (output_full),
16// descriptor-open failure, and errors outside the POSIX conversion model.
17// The OS cannot distinguish WHY a byte sequence is invalid, only that it is.
40enum class iconv_error {
41 invalid_sequence,
42 incomplete_sequence,
43 output_full,
44 open_failed,

46};

48// error_mode — how a view reports a codec failure, as a template
49// parameter rather than as a separate class template. Modelled on P2728's
50// `to_utf_view_error_kind` (see docs/p2728-alignment.md): an enum rather than a
51// bool so that kinds beyond these two can be added, such as one that also
52// surfaces the offending input bytes.
53//
54// replacement — substitute U+FFFD on decode and for ill-formed UTF-32;
55// substitute '?' when an encoder cannot represent a scalar
56// report — the value type becomes expected<T, whatwg::error>
71enum class error_mode {
72 replacement,
