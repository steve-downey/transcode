42constexpr std::optional<encoding> get_encoding(std::text_encoding::id id) noexcept {
43 if (id == std::text_encoding::id::other || id == std::text_encoding::id::unknown)
44 return std::nullopt;
45 const std::text_encoding registered{id};
46 if (auto r = get_encoding(std::string_view{registered.name()}))
47 return r;
48 for (const char* alias : registered.aliases()) {
49 if (auto r = get_encoding(std::string_view{alias}))
50 return r;
51 }
52 return std::nullopt;
53}

62constexpr std::optional<std::text_encoding::id> to_text_encoding_id(encoding e) noexcept {
63 if (e == encoding::replacement || e == encoding::x_user_defined)
