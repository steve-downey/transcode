3// The content transfer encodings a MIME reader undoes, as lazy views of bytes.
4//
5// body | base64_decode RFC 2045 section 6.8
6// body | quoted_printable_decode RFC 2045 section 6.7
7// word | q_decode RFC 2047 section 4.2, "Q" encoded-text
8//
9// Each is a forward range of `char` that decodes as it is iterated, so a part
10// can be handed to a decoder without first being copied. Their input is the
11// contiguous bytes of the message, since a body is a slice of it.
12//
13// Both are lenient, as readers are. Base64 skips whatever is not in the
14// alphabet (line breaks, mostly) and stops at padding. Quoted-printable keeps
15// a stray "=" as it is.
16

73// The two quoted-printable dialects. `body` is RFC 2045: "=" at the end of a
74// line is a soft line break, and white space at the end of a line is padding
75// added in transit. `encoded_word` is RFC 2047's "Q": "_" is a space, and
76// there are no lines.
77enum class quoted_printable_dialect { body, encoded_word };

148// The bytes of a decoded range, in a string.
149template <std::ranges::input_range R>
150std::string collect_bytes(R&& decoded) {
151 std::string bytes;
152 for (char c : decoded) {
153 bytes.push_back(c);
154 }
155 return bytes;
156}
