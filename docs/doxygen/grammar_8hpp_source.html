3// The syntax a MIME reader needs, written close to the ABNF of the RFCs that
4// define it. Every entry point parses a whole input and yields a value or the
5// offset and description of the first thing that was wrong.
6//
7// RFC 5322 header fields, with folding
8// RFC 2045 Content-Type and Content-Transfer-Encoding
9// RFC 2046 multipart bodies
10// RFC 2047 encoded-words in unstructured header text
11//
12// Deliberately left out: RFC 2231 parameter continuations and charset
13// extensions; comments and other CFWS (only white space and folding);
14// obsolete syntax; structured header fields beyond Content-Type and
15// Content-Transfer-Encoding, so an address list is read as unstructured text.
16//
17// Line endings are CRLF, as the RFCs say, or a bare LF, as files have them.

68// A word of unstructured header text: white space, plain text, or an
69// encoded-word (RFC 2047) with its charset and its still-encoded text.
70struct header_chunk {
71 enum class kind { space, text, encoded_word };
72
73 kind what = kind::text;
74 std::string text; // the bytes for space and text; the encoded-text of an encoded-word
75 std::string charset; // encoded-word only
76 char encoding = 'Q'; // 'B' or 'Q', encoded-word only
77
78 // The encoded-word as it was written, for a reader that cannot decode it.
79 std::string spelling() const;
80};
