3// The eager bulk helpers: decode_to, encode_to, decode_into, encode_into,
4// in the form that takes the codec as an argument. The form that names a
5// WHATWG encoding as a template parameter is in
6// <beman/transcode/whatwg/bulk.hpp>.
7//
8// Every one is a one-line forward to the pipeline it names. That is the whole
9// of the implementation and it is also the whole of the specification: the
10// wording says "expression-equivalent to" and shows the pipeline, so there is
11// no second description of decoding to keep in step with the first.
12//
13// They are proposed. `ranges::to` and `ranges::copy` already do the job, and
14// a name that saves no work still saves the reader looking for it -- see
15// papers/transcode-view.md, "Bulk conversion to owned storage".

67template <text_decoder Codec, legacy_byte_range R>
68constexpr std::vector<char32_t> decode_to(Codec codec, R&& source) {
69 return std::forward<R>(source) | decode(codec) | std::ranges::to<std::vector<char32_t>>();
70}
71
73template <text_encoder Codec, typename Container, unicode_scalar_range R>
74constexpr Container encode_to(Codec codec, R&& source) {
75 return std::forward<R>(source) | encode(codec) | std::ranges::to<Container>();
76}
77
79template <text_decoder Codec, legacy_byte_range R, std::output_iterator<char32_t> Output>
80constexpr void decode_into(Codec codec, R&& source, Output output) {
81 std::ranges::copy(std::forward<R>(source) | decode(codec), output);
82}
83
85template <text_encoder Codec, unicode_scalar_range R, std::output_iterator<char> Output>
86constexpr void encode_into(Codec codec, R&& source, Output output) {
87 std::ranges::copy(std::forward<R>(source) | encode(codec), output);
88}
