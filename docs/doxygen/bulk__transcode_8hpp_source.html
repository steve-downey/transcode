3// The eager bulk helpers: decode_to, encode_to, decode_into, encode_into, and
4// the all-or-first-error forms decode_to_or_error and encode_to_or_error, in
5// the form that takes the codec as an argument. The form that names a WHATWG
6// encoding as a template parameter is in <beman/transcode/whatwg/bulk.hpp>.
7//
8// Every one is a one-line forward to the pipeline it names. That is the whole
9// of the implementation and it is also the whole of the specification: the
10// wording says "expression-equivalent to" and shows the pipeline, so there is
11// no second description of decoding to keep in step with the first.
12//
13// They are proposed. `ranges::to` and `ranges::copy` already do the job, and
14// a name that saves no work still saves the reader looking for it -- see
15// papers/transcode-view.md, "Bulk conversion to owned storage".
16

86template <legacy_byte_range R, text_decoder Codec>
87constexpr std::vector<char32_t> decode_to(R&& source, Codec codec) {
88 return std::forward<R>(source) | decode(codec) | std::ranges::to<std::vector<char32_t>>();
89}
90
92template <typename Container, unicode_scalar_range R, text_encoder Codec>
93constexpr Container encode_to(R&& source, Codec codec) {
94 return std::forward<R>(source) | encode(codec) | std::ranges::to<Container>();
95}

98template <legacy_byte_range R, text_decoder Codec, std::output_iterator<char32_t> Output>
99constexpr void decode_into(R&& source, Codec codec, Output output) {
100 std::ranges::copy(std::forward<R>(source) | decode(codec), output);
101}

104template <unicode_scalar_range R, text_encoder Codec, std::output_iterator<char> Output>
105constexpr void encode_into(R&& source, Codec codec, Output output) {
106 std::ranges::copy(std::forward<R>(source) | encode(codec), output);
