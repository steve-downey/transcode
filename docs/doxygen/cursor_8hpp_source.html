3// A parser-combinator kit for the MIME reader example, copied from specgen
4// (include/beman/specgen/parse/cursor.hpp, the same author and license), which
5// adapted it from compile-time-scheme (src/smd/smdscheme/parser/cursor.hpp).

7// Changes from specgen's copy: the namespace is example-local; the line and
8// column tracking is gone, since a MIME message is bytes and not lines of
9// source text; the cursor reads raw bytes, and `char` is only the byte's
10// spelling here. Nothing is interpreted as text until a part's charset says
11// how.

21// An immutable view of the bytes not yet consumed, with the offset of the
22// first of them in the original input. Advancing returns a new cursor, so a
23// parser can checkpoint and backtrack by keeping the old one.
24class cursor {
25 public:
26 constexpr explicit cursor(std::string_view bytes);
27
28 constexpr bool empty() const;
29 constexpr char peek() const; // precondition: !empty()
30 constexpr cursor bump() const; // precondition: !empty()
31 constexpr std::size_t offset() const; // offset from the start of the input
32 constexpr std::string_view remaining() const; // the bytes not yet consumed
33
34 // A cursor that has consumed the first `n` bytes of `remaining()`.
35 constexpr cursor advance(std::size_t n) const;
36
37 private:
38 std::string_view input_;
39 std::size_t offset_ = 0;
40};
41
42constexpr cursor::cursor(std::string_view bytes) : input_{bytes} {}
43
44constexpr bool cursor::empty() const { return input_.empty(); }
45
46constexpr char cursor::peek() const { return input_.front(); }
47
48constexpr cursor cursor::bump() const { return advance(1); }
49
50constexpr cursor cursor::advance(std::size_t n) const {
51 cursor next{*this};
52 next.input_.remove_prefix(n);
53 next.offset_ += n;
54 return next;
55}
56
57constexpr std::size_t cursor::offset() const { return offset_; }
58
59constexpr std::string_view cursor::remaining() const { return input_; }
