203template <encode_codec Codec, std::ranges::input_range R, transcode_error_kind E>
204 requires unicode_scalar_range<R>
205constexpr void encode_view<Codec, R, E>::iterator::load() {
206 if (current_ == end_) {
207 done_ = true;
208 return;
209 }
210 auto r = codec_.encode_one(*current_++);
211 if (r.is_error) {
212 is_error_ = true;
213 len_ = 1;
214 } else {
215 is_error_ = false;
216 for (int i = 0; i < r.count; ++i)
217 buf_[static_cast<std::size_t>(i)] = r.bytes[static_cast<std::size_t>(i)];
218 len_ = r.count;
219 }
220 pos_ = 0;
221}

223template <encode_codec Codec, std::ranges::input_range R, transcode_error_kind E>
224 requires unicode_scalar_range<R>
227constexpr auto encode_view<Codec, R, E>::iterator::operator*() const -> result_t {
228 if (is_error_) {
229 // The lossy encoder substitutes '?'; the expected one reports why.
230 if constexpr (E == transcode_error_kind::expected)
231 return result_t(std::unexpect, whatwg_error::unmapped_codepoint);
232 else
233 return '?';
234 }
235 return static_cast<char>(buf_[static_cast<std::size_t>(pos_)]);
236}
