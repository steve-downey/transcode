161template <encode_codec Codec, std::ranges::input_range R, transcode_error_kind E>
162 requires unicode_scalar_range<R>
163constexpr void encode_view<Codec, R, E>::iterator::load() {
164 if (current_ == end_) {
165 done_ = true;
166 return;
167 }
168 auto r = codec_.encode_one(*current_++);
169 if (r.is_error) {
170 is_error_ = true;
171 len_ = 1;
172 } else {
173 is_error_ = false;
174 for (int i = 0; i < r.count; ++i)
175 buf_[static_cast<std::size_t>(i)] = r.bytes[static_cast<std::size_t>(i)];
176 len_ = r.count;
177 }
178 pos_ = 0;
179}

181template <encode_codec Codec, std::ranges::input_range R, transcode_error_kind E>
182 requires unicode_scalar_range<R>
183constexpr auto encode_view<Codec, R, E>::iterator::operator*() const -> result_t {
184 if (is_error_) {
185 // The lossy encoder substitutes '?'; the expected one reports why.
186 if constexpr (E == transcode_error_kind::expected)
187 return result_t(std::unexpect, whatwg_error::unmapped_codepoint);
188 else
189 return '?';
190 }
191 return static_cast<char>(buf_[static_cast<std::size_t>(pos_)]);
192}
